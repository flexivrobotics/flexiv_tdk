66void ReadDigitalInputTask(flexiv::tdk::TransparentCartesianTeleopLAN& teleop)
67{
68 while (g_running.load() && !teleop.any_fault()) {
69 try {
70 teleop.Engage(0, teleop.digital_inputs(0).first[0]);
71 } catch (const std::exception& e) {
72 spdlog::error("Exception in ReadDigitalInputTask: {}", e.what());
73 }
74 std::this_thread::sleep_for(std::chrono::milliseconds(10));
75 }
76 spdlog::info("ReadDigitalInputTask exiting.");
77 return;

291int main(int argc, char* argv[])
292{
293 std::string follower_sn {};
294 std::string leader_sn {};
295 std::string whitelist_ip {};
296 std::vector<std::string> network_interface_whitelist {};
297 int opt = 0;
298 int longIndex = 0;
299 bool enable_digital_input = false;
300
301 while ((opt = getopt_long_only(argc, argv, "f:l:i:D", kLongOptions, &longIndex)) != -1) {
302 switch (opt) {
303 case 'f':
304 follower_sn = std::string(optarg);
305 break;
306 case 'l':
307 leader_sn = std::string(optarg);
308 break;
309 case 'i':
310 whitelist_ip = std::string(optarg);
311 network_interface_whitelist.emplace_back(whitelist_ip);
312 break;
313 case 'D':
314 enable_digital_input = true;
315 break;
316 default:
317 PrintHelp();
318 return 1;
319 }
320 }
321 if (leader_sn.empty() || follower_sn.empty()) {
322 PrintHelp();
323 return 1;
324 }
325 if (network_interface_whitelist.empty()) {
326 spdlog::warn(
327 "network_interface_whitelist is not provided, will search all network interfaces.");
328 }
329
330 try {
331
332 // Create teleop control interface
333 flexiv::tdk::TransparentCartesianTeleopLAN tctl(
334 {{leader_sn, follower_sn}}, network_interface_whitelist);
335
336 // Run initialization sequence
337 tctl.Init();
338
339 // Start control loop
340 tctl.Start();
341
342 // Start console and pedal input threads
343 std::thread console_thread(std::bind(ConsoleTask, std::ref(tctl)));
344
345 // Start pedal_thread based on the new flag or the role
346 std::optional<std::thread> pedal_thread;
347
348 if (enable_digital_input) {
349 spdlog::info("Starting ReadDigitalInputTask thread.");
350 pedal_thread.emplace(ReadDigitalInputTask, std::ref(tctl));
351 } else {
352 spdlog::info("ReadDigitalInputTask thread NOT started (-D flag not provided).");
353 }
354
355 // Wait for threads to finish
356 console_thread.join();
357
358 // Stop all threads, notify other threads exit
359 g_running.store(false);
360
361 // Wait for digital reading task exit
362 if (pedal_thread && pedal_thread->joinable()) {
363 pedal_thread->join();
364 }
365
366 // Exit high transparency teleop
367 tctl.Stop();
368
369 } catch (const std::exception& e) {
370 spdlog::error(e.what());
371 return 1;
372 }
373
374 return 0;
375}
