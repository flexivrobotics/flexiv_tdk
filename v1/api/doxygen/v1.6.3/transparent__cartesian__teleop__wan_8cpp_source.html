45void PrintTeleopStatus(unsigned int idx, const flexiv::tdk::TeleopStatus& status)
46{
47 spdlog::info(
48 "Teleop pair {} status: initialized={} started={} engaged={} stopped={} fault={} "
49 "motion_restricted={} latency={:.1f}/{:.1f} ms",
50 idx, status.initialized, status.started, status.engaged, status.stopped, status.fault,
51 status.motion_restricted, status.latency_ms, status.latency_threshold_ms);
52 if (status.primary.code == flexiv::tdk::TeleopIssueCode::NONE) {
53 spdlog::info("No teleop restriction. Safe to continue.");
54 return;
55 }
56 const auto& issue = status.primary;
57 spdlog::warn("[{}/{}] {} | {} | {}",
58 flexiv::tdk::TeleopIssueLevelStr[static_cast<size_t>(issue.level)],
59 flexiv::tdk::TeleopIssueSideStr[static_cast<size_t>(issue.side)], issue.title,
60 issue.description, issue.suggestion);
61 for (const auto& extra : status.issues) {
62 if (extra.code == issue.code && extra.side == issue.side
63 && extra.joint_index == issue.joint_index) {
64 continue;
65 }
66 spdlog::warn(" also: [{}/{}] {}",
67 flexiv::tdk::TeleopIssueCodeStr[static_cast<size_t>(extra.code)],
68 flexiv::tdk::TeleopIssueSideStr[static_cast<size_t>(extra.side)], extra.title);
69 }
70}

107void ReadDigitalInputTask(flexiv::tdk::TransparentCartesianTeleopWAN& teleop)
108{
109 bool logged_idle = false;
110 std::string last_error;
111 while (g_running.load()) {
112 try {
113 const auto status = teleop.GetTeleopStatus(0);
114 // Robot fault / Stop() drops teleop back to not-started. Engage() is
115 // invalid in that state and must not be polled every cycle.
116 if (!status.started || status.stopped) {
117 if (!logged_idle) {
118 spdlog::warn(
119 "ReadDigitalInputTask: teleop is not started, pause Engage "
120 "(call Init + Start to resume)");
121 logged_idle = true;
122 }
123 } else {
124 logged_idle = false;
125 teleop.Engage(0, teleop.digital_inputs(0)[0]);
126 }
127 last_error.clear();
128 } catch (const std::exception& e) {
129 if (last_error != e.what()) {
130 spdlog::error("Exception in ReadDigitalInputTask: {}", e.what());
131 last_error = e.what();
132 }
133 }
134 std::this_thread::sleep_for(std::chrono::milliseconds(10));
135 }
136 spdlog::info("ReadDigitalInputTask exiting.");
137 return;
138}

143void ConsoleTask(flexiv::tdk::TransparentCartesianTeleopWAN& teleop)
144{
145 auto PrintCommandMenu = []() {
146 std::cout << R"(
147 --- Axis Lock (leader only) ---
148 x/y/z : Toggle translation lock in WORLD coord (X/Y/Z)
149 q/w/e : Toggle orientation lock in WORLD coord (Rx/Ry/Rz)
150 X/Y/Z : Toggle translation lock in TCP coord (X/Y/Z)
151 Q/W/E : Toggle orientation lock in TCP coord (Rx/Ry/Rz)
152 a : Unlock all axes (TCP coord)
153 A : Lock all axes (TCP coord)
154
155 --- Teleop Engagement ---
156 r : Engage teleop
157 R : Disengage teleop
158
159 --- Null Space Posture ---
160 i/I : Set local nullspace to Preferred/Home posture
161
162 --- Max Contact Wrench ---
163 p : Set default max contact wrench
164
165 --- Reinit and start ---
166 u : Recall Init and Start
167 U : Stop teleop
168
169 --- Tcp message latency ---
170 l : print current message latency (disconnected / clock mismatch / over limit)
171
172 --- Teleop status ---
173 h : print why teleop is restricted / paused and what to do next
174
175 --- Identity ---
176 n : print role() and robot_pair_sn()
177
178 --- Help ---
179 Any other key to show this help menu
180 )" << std::endl;
181 };
182
183 unsigned int index = 0;
184 const bool is_leader = (teleop.role() == flexiv::tdk::Role::WAN_TELEOP_LEADER);
185 flexiv::tdk::AxisLock cmd;
186 if (is_leader) {
187 teleop.GetAxisLockState(index, cmd);
188 }
189
190 auto apply_axis_lock = [&]() {
191 if (!is_leader) {
192 spdlog::warn("Axis lock is only available on the leader");
193 return;
194 }
195 teleop.SetAxisLockCmd(index, cmd);
196 };
197
198 while (g_running.load() && !teleop.fault(0)) {
199
200 std::string user_input {};
201
202 std::getline(std::cin, user_input);
203
204 if (user_input.empty()) {
205 spdlog::warn("Empty command!");
206 PrintCommandMenu();
207 continue;
208 }
209
210 try {
211 switch (user_input[0]) {
212 case 'x':
213 cmd.lock_trans_axis[0] = !cmd.lock_trans_axis[0];
214 cmd.coord = flexiv::tdk::CoordType::COORD_WORLD;
215 apply_axis_lock();
216 break;
217 case 'y':
218 cmd.lock_trans_axis[1] = !cmd.lock_trans_axis[1];
219 cmd.coord = flexiv::tdk::CoordType::COORD_WORLD;
220 apply_axis_lock();
221 break;
222 case 'z':
223 cmd.lock_trans_axis[2] = !cmd.lock_trans_axis[2];
224 cmd.coord = flexiv::tdk::CoordType::COORD_WORLD;
225 apply_axis_lock();
226 break;
227 case 'q':
228 cmd.lock_ori_axis[0] = !cmd.lock_ori_axis[0];
229 cmd.coord = flexiv::tdk::CoordType::COORD_WORLD;
230 apply_axis_lock();
231 break;
232 case 'w':
233 cmd.lock_ori_axis[1] = !cmd.lock_ori_axis[1];
234 cmd.coord = flexiv::tdk::CoordType::COORD_WORLD;
235 apply_axis_lock();
236 break;
237 case 'e':
238 cmd.lock_ori_axis[2] = !cmd.lock_ori_axis[2];
239 cmd.coord = flexiv::tdk::CoordType::COORD_WORLD;
240 apply_axis_lock();
241 break;
242 case 'X':
243 cmd.lock_trans_axis[0] = !cmd.lock_trans_axis[0];
244 cmd.coord = flexiv::tdk::CoordType::COORD_TCP;
245 apply_axis_lock();
246 break;
247 case 'Y':
248 cmd.lock_trans_axis[1] = !cmd.lock_trans_axis[1];
249 cmd.coord = flexiv::tdk::CoordType::COORD_TCP;
250 apply_axis_lock();
251 break;
252 case 'Z':
253 cmd.lock_trans_axis[2] = !cmd.lock_trans_axis[2];
254 cmd.coord = flexiv::tdk::CoordType::COORD_TCP;
255 apply_axis_lock();
256 break;
257 case 'Q':
258 cmd.lock_ori_axis[0] = !cmd.lock_ori_axis[0];
259 cmd.coord = flexiv::tdk::CoordType::COORD_TCP;
260 apply_axis_lock();
261 break;
262 case 'W':
263 cmd.lock_ori_axis[1] = !cmd.lock_ori_axis[1];
264 cmd.coord = flexiv::tdk::CoordType::COORD_TCP;
265 apply_axis_lock();
266 break;
267 case 'E':
268 cmd.lock_ori_axis[2] = !cmd.lock_ori_axis[2];
269 cmd.coord = flexiv::tdk::CoordType::COORD_TCP;
270 apply_axis_lock();
271 break;
272 case 'a':
273 cmd.lock_ori_axis = {false, false, false};
274 cmd.lock_trans_axis = {false, false, false};
275 cmd.coord = flexiv::tdk::CoordType::COORD_TCP;
276 apply_axis_lock();
277 break;
278 case 'A':
279 cmd.lock_ori_axis = {true, true, true};
280 cmd.lock_trans_axis = {true, true, true};
281 cmd.coord = flexiv::tdk::CoordType::COORD_TCP;
282 apply_axis_lock();
283 break;
284 case 'r':
285 teleop.Engage(0, true);
286 break;
287 case 'R':
288 teleop.Engage(0, false);
289 break;
290 case 'i':
291 teleop.SetNullSpacePosture(0, kPreferredJntPos);
292 break;
293 case 'I':
294 teleop.SetNullSpacePosture(0, kHomeJntPos);
295 break;
296 case 'p':
297 teleop.SetMaxContactWrench(0, kDefaultMaxContactWrench);
298 break;
299 case 'u':
300 teleop.Init();
301 teleop.Start();
302 break;
303 case 'U':
304 teleop.Stop();
305 break;
306 case 'l': {
307 double latency_ms {};
308 const bool ok = teleop.CheckTeleopConnectionLatency(0, latency_ms);
309 if (ok) {
310 spdlog::info("pair 0 message latency: {} ms (within limit)", latency_ms);
311 } else if (latency_ms < 0.0) {
312 spdlog::warn("pair 0 clock mismatch: latency {} ms", latency_ms);
313 } else if (latency_ms > 1.0e12) {
314 spdlog::warn("pair 0 disconnected: latency {} ms", latency_ms);
315 } else {
316 spdlog::warn("pair 0 latency over limit: {} ms", latency_ms);
317 }
318 break;
319 }
320 case 'h':
321 PrintTeleopStatus(0, teleop.GetTeleopStatus(0));
322 break;
323 case 'n': {
324 const auto pair = teleop.robot_pair_sn(0);
325 spdlog::info("pair 0 role={} leader_sn={} follower_sn={}",
326 flexiv::tdk::RoleTypeStr[static_cast<size_t>(teleop.role())], pair.first,
327 pair.second);
328 break;
329 }
330
331 default:
332 spdlog::warn("Invalid command!");
333 PrintCommandMenu();
334 break;
335 }
336 } catch (const std::exception& e) {
337 spdlog::error("Exception in ConsoleTask: {}", e.what());
338 g_running.store(false);
339 return;
340 }
341 }
342 spdlog::info("Console thread exiting.");
343 return;
344}

346int main(int argc, char* argv[])
347{
348 std::string follower_sn, leader_sn, teleop_role, tcp_role, public_server_ip, lan_ip, wan_iface;
349 unsigned int server_port = 0;
350 std::vector<std::string> lan_interface_whitelist {};
351 std::vector<std::string> wan_interface_whitelist {};
352 bool enable_digital_input = false;
353
354 int opt = 0;
355 while ((opt = getopt_long_only(argc, argv, "l:f:r:t:i:p:A:W:D", kLongOptions, nullptr)) != -1) {
356 switch (opt) {
357 case 'f':
358 follower_sn = std::string(optarg);
359 break;
360 case 'l':
361 leader_sn = std::string(optarg);
362 break;
363 case 'r':
364 teleop_role = std::string(optarg);
365 break;
366 case 't':
367 tcp_role = std::string(optarg);
368 break;
369 case 'i':
370 public_server_ip = std::string(optarg);
371 break;
372 case 'p':
373 try {
374 server_port = std::stoi(optarg);
375 } catch (...) {
376 spdlog::error("Invalid port number: {}", optarg);
377 return 1;
378 }
379 break;
380 case 'A':
381 lan_ip = std::string(optarg);
382 lan_interface_whitelist.push_back(lan_ip);
383 break;
384 case 'W':
385 wan_iface = std::string(optarg);
386 wan_interface_whitelist.push_back(wan_iface);
387 break;
388 case 'D':
389 enable_digital_input = true;
390 break;
391 default:
392 PrintHelp();
393 return 1;
394 }
395 }
396 if (follower_sn.empty() || leader_sn.empty() || tcp_role.empty() || teleop_role.empty()
397 || public_server_ip.empty() || server_port == 0) {
398 PrintHelp();
399 return 1;
400 }
401 if (lan_interface_whitelist.empty()) {
402 spdlog::warn("LAN whitelist is not provided, will search all network interfaces.");
403 }
404 if (wan_interface_whitelist.empty()) {
405 spdlog::warn(
406 "WAN interface whitelist is not provided, will search all network interfaces.");
407 }
408
409 // Whether this is a TCP server or client
410 bool is_tcp_server;
411 if (tcp_role == "server") {
412 is_tcp_server = true;
413 } else if (tcp_role == "client") {
414 is_tcp_server = false;
415 } else {
416 spdlog::error("Valid inputs for [-t] are: server, client");
417 return 1;
418 }
419
420 // Whether this is leader or follower
421 if (teleop_role == "follower") {
422 kRole = flexiv::tdk::Role::WAN_TELEOP_FOLLOWER;
423 } else if (teleop_role == "leader") {
424 kRole = flexiv::tdk::Role::WAN_TELEOP_LEADER;
425 } else {
426 spdlog::error("Valid inputs for [-r] are: follower, leader");
427 return 1;
428 }
429
430 // Network configuration
431 flexiv::tdk::NetworkCfgStd network_cfg;
432 network_cfg.is_tcp_server = is_tcp_server;
433 network_cfg.public_ipv4_address = public_server_ip;
434 network_cfg.listening_port = server_port;
435 network_cfg.lan_interface_whitelist = lan_interface_whitelist;
436 network_cfg.wan_interface_whitelist = wan_interface_whitelist;
437
438 std::vector<std::pair<std::string, std::string>> robot_sn_pairs {};
439 robot_sn_pairs.push_back({leader_sn, follower_sn});
440 try {
441
442 // Allocate tdk object (Standard Edition TCP peer-to-peer)
443 flexiv::tdk::TransparentCartesianTeleopWAN tctw(robot_sn_pairs, kRole, network_cfg);
444 const auto pair_sn = tctw.robot_pair_sn(0);
445 spdlog::info("This instance role={} leader_sn={} follower_sn={}",
446 flexiv::tdk::RoleTypeStr[static_cast<size_t>(tctw.role())], pair_sn.first,
447 pair_sn.second);
448
449 // Init high transparency teleop
450 tctw.Init();
451
452 // Start high transparency teleop
453 tctw.Start();
454
455 // Set max contact wrench
456 tctw.SetMaxContactWrench(0, kDefaultMaxContactWrench);
457
458 // Start console_thread
459 std::thread console_thread(std::bind(ConsoleTask, std::ref(tctw)));
460
461 // Start pedal_thread based on the new flag or the role
462 std::optional<std::thread> pedal_thread;
463 // Modified condition: Start if -D is given OR if role is leader (original behavior)
464 if (teleop_role == "leader" && enable_digital_input) {
465 spdlog::info(
466 "Starting ReadDigitalInputTask thread as role is 'leader' and requested by -D "
467 "flag.");
468 pedal_thread.emplace(ReadDigitalInputTask, std::ref(tctw));
469 } else {
470 spdlog::info(
471 "ReadDigitalInputTask thread NOT started (role is not 'leader' or -D flag not "
472 "provided).");
473 }
474
475 // Wait for console_thread to finish
476 console_thread.join();
477
478 // Stop all threads, notify other threads exit
479 g_running = false;
480
481 // Wait for digital reading task exit
482 if (pedal_thread && pedal_thread->joinable()) {
483 pedal_thread->join();
484 }
485
486 // Exit high transparency teleop
487 tctw.Stop();
488
489 } catch (const std::exception& e) {
490 spdlog::error(e.what());
491 return 1;
492 }
493
494 return 0;
495}
